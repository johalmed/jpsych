cmake_minimum_required(VERSION 3.22)
project(jAudioMathMIDI VERSION 0.1.0)
include(FetchContent)
FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(juce)
juce_add_plugin(jAudioMathMIDI COMPANY_NAME "jAudio" BUNDLE_ID "com.jaudio.mathmidi" PLUGIN_MANUFACTURER_CODE jAud PLUGIN_CODE JmMd FORMATS VST3 PRODUCT_NAME "jAudio Math MIDI" IS_MIDI_EFFECT TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT TRUE IS_SYNTH FALSE COPY_PLUGIN_AFTER_BUILD TRUE)
target_sources(jAudioMathMIDI PRIVATE src/PluginProcessor.cpp src/PluginProcessor.h src/PluginEditor.cpp src/PluginEditor.h src/Formula.h)
target_compile_features(jAudioMathMIDI PRIVATE cxx_std_17)
target_link_libraries(jAudioMathMIDI PRIVATE juce::juce_audio_utils juce::juce_audio_processors juce::juce_dsp)
target_compile_definitions(jAudioMathMIDI PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
